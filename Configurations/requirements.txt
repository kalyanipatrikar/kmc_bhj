# AQx-2 transfer integral pipeline - environment
#
# Run order:
#   get_pair_structure.ipynb   2161927.cif  ->  crystal_dimers/*_dimer.xyz
#   generate_configs.ipynb     dimers       ->  configs_<motif>/, 40 frames each
#   transfer_integrals.ipynb   configs      ->  transfer_integrals_<motif>.csv,
#                                               J_<motif>.txt
#
# Python 3.12.10   (3.10-3.12 all work; mace-torch does not yet support 3.13)
#
# These are the versions the pipeline was developed and checked against. The
# pins are there so a new machine reproduces the same numbers; loosen them to
# >= if you hit a resolver conflict.
#
# ---------------------------------------------------------------------------
# Creating the environment
#
#   conda create -n aqx2 python=3.12
#   conda activate aqx2
#   pip install -r requirements.txt
#
# or with venv:
#
#   python3.12 -m venv aqx2 && source aqx2/bin/activate
#   pip install -r requirements.txt
#
# ---------------------------------------------------------------------------
# Two things this file cannot express
#
# 1. GPU torch. The pin below installs whatever build pip defaults to, which
#    is the CPU build on many platforms. Stage 2 is 45 ps of MACE dynamics per
#    motif and is impractically slow on a CPU, so install the CUDA build first,
#    then this file:
#
#      pip install torch==2.5.1 --index-url https://download.pytorch.org/whl/cu124
#
#    Match the cu### suffix to the driver on the machine; check afterwards with
#    python -c "import torch; print(torch.cuda.is_available())"
#
# 2. Platform. pyscf has no Windows build - run stage 3 under WSL, Linux or
#    macOS. Stages 1 and 2 are fine on Windows. Nothing else here is platform
#    specific.
# ---------------------------------------------------------------------------

# stage 1: crystal structure -> dimers
numpy==2.5.2
scipy==1.17.1

# stage 2: MACE molecular dynamics
torch==2.5.1
ase==3.29.0
mace-torch==0.3.16
e3nn==0.4.4          # pinned by mace-torch; listed so the resolver is explicit

# stage 3: DFT, dimer projection
pyscf==2.14.0

# running the notebooks
notebook==7.6.0
ipykernel==7.3.0
